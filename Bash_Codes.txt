# Running containers
docker ps

# All containers
docker ps -a

# Stop container
docker stop <container_id>

# Remove container
docker rm <container_id>

# Remove image
docker rmi voyage-analytics

# Run container again
docker run -p 5000:5000 voyage-analytics

# Streamlit
streamlit run streamlit_app/streamlit_app.py

# API
python api/app.py

# Docker build
docker build -t voyage-api -f api/Dockerfile .

# Mlflow Run
mlflow ui --backend-store-uri sqlite:///mlflow/mlflow.db --default-artifact-root ./mlflow/artifacts --host 0.0.0.0 --port 5000

# Docker scheduler
docker exec -it airflow_scheduler bash
