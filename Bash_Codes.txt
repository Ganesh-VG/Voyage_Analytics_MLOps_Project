# Running containers
docker ps

# All containers
docker ps -a

# Stop container
docker stop <container_id>

# Remove container
docker rm <container_id>

# Remove image
docker rmi voyage-analytics

# Run container again
docker run -p 5000:5000 voyage-analytics

# Streamlit
streamlit run streamlit_app/streamlit_app.py

# API
python api/app.py

# Docker build
docker build -t voyage-api -f api/Dockerfile .

# Stop Docker
docker compose down

# Mlflow Run
mlflow ui --backend-store-uri sqlite:///mlflow/mlflow.db --default-artifact-root ./mlflow/artifacts --host 0.0.0.0 --port 5000

# Docker scheduler
docker exec -it airflow_scheduler bash

# Run airflow
cd airflow

docker compose up -d

Open http://localhost:8080

# Checking the Containers
docker compose ps

# Run Jenkins - first set the directory to jenkins
docker compose build
docker compose up -d

# Use the below code to generate the password in jenkins for the first time
docker exec voyage_jenkins sh -c "cat /var/jenkins_home/secrets/initialAdminPassword"

# Code to open the bash inside docker
docker exec -it voyage_jenkins bash

# another way to to generate the password in jenkins
cat /var/jenkins_home/secrets/initialAdminPassword
